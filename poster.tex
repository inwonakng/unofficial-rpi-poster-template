\begin{pframe}

	\begin{pcolumn}{\colwidth}

		\begin{pblock}{Layout commands}{24cm}

			These helpers make poster geometry explicit.

			\begin{itemize}
				\item \texttt{pframe}: one Beamer frame plus left/right gutters.
				\item \texttt{pcolumn\{width\}}: a top-aligned minipage of exact width.
				\item \texttt{prow}: puts columns on one row; use it for nested columns.
				\item \texttt{pblock[style]\{title\}\{height\}}: block with a target height.
			\end{itemize}

			Use \texttt{alert} or \texttt{example} as optional block styles. If content is taller than the target, the block grows instead of overprinting the next block.

			\heading{Usual skeleton}
			\texttt{pframe $\rightarrow$ pcolumn $\rightarrow$ pblock}\par
			Nested layout: \texttt{prow $\rightarrow$ pcolumn $\rightarrow$ pblock}.

		\end{pblock}

		\begin{pblock}{Width recipe}{12.5cm}

			Pick the number of columns first, then solve for widths.

			\begin{itemize}
				\item Top level: $(N+1)s + Nw = \texttt{\textbackslash textwidth}$.
				\item Nested row: $\sum w_i + (N-1)s = \texttt{parent width}$.
				\item Here: $s=\texttt{\textbackslash pcolumnsepwidth}$ and $w=\texttt{\textbackslash colwidth}$.
			\end{itemize}

		\end{pblock}

		\begin{pblock}[alert]{Why not Beamer columns?}{%
				\textheight-24cm-12.5cm-2\blocksepheight
			}

			Beamer's \texttt{columns} is made for slides, not grid-filling posters.

			\begin{itemize}
				\item It adds internal spacing, so the visible width is not just the number you typed.
				\item Column height follows content; it does not reserve an exact block height.
				\item Nested layouts are harder to make flush with the outer grid.
			\end{itemize}

			The custom helpers use plain minipages and fixed gutters, so the math stays predictable.

		\end{pblock}

	\end{pcolumn}

	\begin{pcolumn}{\twocolwidth}

		\begin{pblock}{Height recipe}{24.00cm}

			Give every block a target height and let the last block fill the remainder.

			\[
				\sum h_i + (M-1)\,\texttt{\textbackslash blocksepheight} = \texttt{\textbackslash textheight}
			\]

			For a column with blocks of \texttt{24cm}, \texttt{12.5cm}, and one remaining block:

			\[
				\texttt{\textbackslash textheight - 24cm - 12.5cm - 2\textbackslash blocksepheight}
			\]

			\begin{table}
				\centering
				\begin{tabular}{l l}
					\toprule
					\textbf{Use} & \textbf{Pattern} \\
					\midrule
					Same-height stack & choose fixed \texttt{cm} values \\
					Fill leftover space & subtract used heights from \texttt{\textbackslash textheight} \\
					Two-column span & \texttt{2\textbackslash colwidth + \textbackslash pcolumnsepwidth} \\
					Too much content & block grows; shorten text or increase height \\
					\bottomrule
				\end{tabular}
			\end{table}

			Keep these formulas near the length definitions in the preamble.

		\end{pblock}

		\begin{prow}
			\begin{pcolumn}{\colwidth}

				\begin{pblock}{Changing type size}{15cm}

					Global sizes live in \texttt{beamerthemerpi.sty}.

					\begin{itemize}
						\item \texttt{block title}: usually \texttt{\textbackslash Large} and bold.
						\item \texttt{block body}: usually \texttt{\textbackslash large}.
						\item \texttt{headline title}: usually \texttt{\textbackslash Huge}.
					\end{itemize}

					Local override: \texttt{\{\textbackslash Large ...\}}. Exact size: \texttt{\textbackslash fontsize\{34\}\{40\}\textbackslash selectfont}.

				\end{pblock}


				\begin{pblock}{Changing separator lengths}{
						\textheight-24cm-15cm-2\blocksepheight
					}

					Two lengths control most whitespace.

					\begin{itemize}
						\item \texttt{\textbackslash pcolumnsepwidth}: horizontal gutter between columns.
						\item \texttt{\textbackslash blocksepheight}: vertical gap between stacked blocks.
					\end{itemize}

					Set them in the preamble:

					\texttt{\textbackslash setlength\{\textbackslash pcolumnsepwidth\}\{0.025\textbackslash textwidth\}}\par
					\texttt{\textbackslash setlength\{\textbackslash blocksepheight\}\{4ex\}}

				\end{pblock}


			\end{pcolumn}

			\begin{pcolumn}{\colwidth}

				\begin{pblock}[example]{Mini layout checklist}{24cm}

					\begin{enumerate}
						\item Choose poster size in \texttt{beamerposter}.
						\item Choose \texttt{\textbackslash pcolumnsepwidth}.
						\item Solve for \texttt{\textbackslash colwidth}.
						\item Set span widths, e.g. \texttt{\textbackslash twocolwidth}.
						\item Pick block heights to sum to \texttt{\textbackslash textheight}.
						\item Build, then trim text or adjust heights.
					\end{enumerate}

					\heading{Rule of thumb}

					Make the grid first, then write to fit it. Clean posters have shared gutters, aligned bottoms, and short text.

				\end{pblock}
